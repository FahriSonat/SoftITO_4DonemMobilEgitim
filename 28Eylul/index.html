const urunleriGetir = async () => { try { console.log("Harici API'ye bağlanılıyor..."); const response = await fetch("https://dummyjson.com/products"); if (!response.ok) { throw new Error(`HTTP Hatası: ${response.status}`); } const data = await response.json(); const vitrin = data.products.slice(0, 5).map((p) => ({ isim: p.title, fiyat: p.price, kategori: p.category, aciklama: p.description, stok: p.stock, etiketler: p.tags, })); console.log(vitrin); } catch (err) { console.warn("API Hatası:", err.message); } }; urunleriGetir();
